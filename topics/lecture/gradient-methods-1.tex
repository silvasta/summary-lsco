\section{Gradient Methods – Part I}

\begin{definition}[Smoothness]
  \(f\) is \(L\)-smooth if \(\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\).
  Consequently,
  \[
    f(y)\le f(x)+\nabla f(x)^\top(y-x)+\tfrac L2\|x-y\|^2.
  \]
\end{definition}

\begin{definition}[Strong convexity]
  \(f\) is \(\mu\)-strongly convex if
  \[
    f(y)\ge f(x)+\nabla f(x)^\top(y-x)+\tfrac\mu2\|y-x\|^2.
  \]
\end{definition}

\subsection{Gradient Descent on Quadratics}
For \(f(x)=\frac12 x^\top Hx+b^\top x\) with \(H\succeq\mu I\) and
\(H\preceq L I\) the iteration becomes
\[
  x_{k+1}-x^*=(I-TH)(x_k-x^*).
\]
The optimal step size \(T^*=2/(L+\mu)\) yields the contraction factor
\[
  \rho=\frac{L-\mu}{L+\mu}=\frac{\kappa-1}{\kappa+1},\qquad\kappa=L/\mu.
\]
Hence \(N\ge(\kappa+1)/2\cdot\log(|x_0-x^*|/\varepsilon)\) iterations suffice.

\subsection{Momentum Methods (Nesterov / Heavy-Ball)}
The two-step recurrence
\[
  q_{k+1}=q_k+Tp_{k+1},\qquad p_{k+1}=(1-2dT)p_k-T\nabla f(q_k+\beta p_k)/L
\]
with the classical parameter choices realises the optimal rate
\[
  \bigl(1-\tfrac1{\sqrt\kappa}\bigr)^N
\]
(up to a constant factor). This rate is optimal among all first-order
methods on the class of smooth strongly-convex functions.
