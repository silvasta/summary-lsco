\section*{1. Convexity (Quiz 1 Tactic)}
\textbf{Basic Tests:}
\begin{itemize}[leftmargin=*]
  \item \textbf{Def:} $f(\theta x + (1-\theta)y) \le \theta f(x) +
    (1-\theta)f(y)$ $\forall \theta \in [0,1]$ $\iff$ epi$(f)$ convex.
  \item \textbf{1st-order:} $f(y) \ge f(x) + \nabla f(x)^\top(y-x)$.
    Monotone grad: $(\nabla f(x)-\nabla f(y))^\top(x-y)\ge0$.
  \item \textbf{2nd-order:} $\nabla^2 f(x) \succeq 0$.
    (\textit{Strict} if $\succ 0$).
  \item \textbf{Preserving:} Non-negative sum, affine comp.
    $f(Ax+b)$, pointwise max $g(x)=\max_i f_i(x)$, intersection of
    sets. $f^{**}=f$ (proper, closed).
\end{itemize}

\textbf{The "NOT CONVEX" Trap List:}
\begin{itemize}[leftmargin=*]
  \item Pointwise \textbf{min} of convex funcs (e.g., $(x+1)^2, (x-1)^2$).
  \item Unique/global min $\nRightarrow$ convex (e.g., $f(x)=|x|$ on
    $[-1,1]$, $1$ outside).
  \item $\nabla^2 f$ bounded below $\nRightarrow$ convex (must be $\ge 0$).
  \item Product $f_1 \cdot f_2$ of convex $\nRightarrow$ convex
    (e.g., $x\cdot(-x)=-x^2$).
  \item $1/f(x)$ or $\exp(f(x))$ where $f$ is convex (not always).
  \item $f(x)=1/\ln x$ on $(1, \infty)$ (Check Hessian: $x=e^2 \to$ neg).
\end{itemize}

\section*{2. Conjugates $f^*(y)$}
\textbf{Definition:} $f^*(y) = \sup_{x} \{y^\top x - f(x)\}$.
\textit{Prop:} $f^{**}=f$ (if proper/closed). $y\in\partial f(x)\iff
x\in\partial f^*(y)$.

\textbf{The 4-Step Recipe:}
1. Maximize: $\nabla_x (y^\top x - f(x)) = 0 \implies \nabla f(x) = y$.
2. Solve for $x$ in terms of $y$ ($x_y$).
3. If $x_y \in \text{int}(\text{dom } f)$, $f^*(y) = y^\top x_y - f(x_y)$.
4. \textbf{Restricted Domain:} If $x \in [a, b]$, check boundaries.
- If $x_y < a$, max is at $x=a$. If $x_y > b$, max is at $x=b$.

\textbf{Standard Pairs \& Quiz Examples:}
\begin{itemize}[leftmargin=*]
  \item $f(x) = -\ln x \to f^*(y) = -1-\ln(-y)$ for $y<0$.
  \item $f(x) = e^x \to f^*(y) = y \ln y - y$ for $y>0$.
  \item $f(x) = \frac{1}{2}x^\top Q x \to f^*(y) = \frac{1}{2}y^\top Q^{-1} y$.
  \item $f(x) = -x^2$ on $[-1,1] \to f^*(y) = 1+|y|$ (V-shape).
  \item $f(x) = x^2+x$ on $[-1,1]$: Stationary $x=\frac{1}{2}(y-1)$.
    $f^*(y) = -y$ ($y\le-1$); $\frac{1}{4}(y-1)^2$ ($-1\le y\le3$);
    $y-2$ ($y\ge3$).
\end{itemize}

\section*{3. Subdifferentials \& Projections}
$\partial f(\bar x)=\{v\mid f(x)\ge f(\bar x)+v^\top(x-\bar x)\}$. \\
\textbf{Optimality:} $0 \in \partial f(x^*) \iff x^*$ global min.
\begin{itemize}[leftmargin=*]
  \item $\partial |x| = \{-1\}$ if $x<0$, $\{1\}$ if $x>0$, and $[-1,
    1]$ if $x=0$.
  \item \textbf{Graphical Tactic:} Minimizer is where the
    subdifferential plot contains the 0-line (look for vertical
    segments spanning $y=0$).
\end{itemize}

\textbf{Projections (PGD):} $x_{k+1} = \text{prox}_{T, C}(x_k - T
\nabla f(x_k))$.
\begin{itemize}[leftmargin=*]
  \item \textbf{$\ell_\infty$-ball ($\|x\|_\infty \le 1$):} $x_i =
    \text{clamp}(z_i, -1, 1)$.
  \item \textbf{$\ell_2$-ball ($\|x\|_2 \le 1$):} $z / \max(1, \|z\|_2)$.
  \item \textbf{$\ell_1$-ball ($\|x\|_1 \le 1$):} Soft-thresholding.
  \item \textbf{Orthogonal $U$:} If $C = \{x \mid \|Ux\|_\infty \le
    1\}$, use $x = U^\top \text{prox}_{\|\cdot\|_\infty}(Uz)$.
\end{itemize}

\section*{4. Duality \& KKT}
\textbf{Primal:} $\min f(x)$ s.t. $g_i(x)\le0$, $h_j(x)=0$. \\
\textbf{Lagrangian:} $\mathcal{L} = f(x) + \sum \lambda_i g_i(x) +
\nu^\top(Ax-b)$. \\
\textbf{Dual Function:} $d(\lambda, \nu) = \inf_x \mathcal{L}$.
\textit{Always} concave. \\
\textbf{Weak Duality:} $d^* \le p^*$ \textit{always}.

\textbf{KKT Conditions} (Strong duality requires Convexity + Slater's
condition: $\exists \hat{x}, g_i(\hat{x}) < 0$):
1. \textbf{Stationarity:} $\nabla_x \mathcal{L}(x^*, \lambda^*, \nu^*) = 0$.
2. \textbf{Primal Feas:} $g(x^*) \le 0, Ax^*=b$.
3. \textbf{Dual Feas:} $\lambda^* \ge 0$.
4. \textbf{Comp. Slackness:} $\lambda_i^* g_i(x^*) = 0$.

\textbf{Tactic:} Write full $\mathcal{L}$, set
$\nabla_x\mathcal{L}=0$, substitute $x$ into $d(\lambda)$, maximize
(often quadratic). Assume constraints active/inactive based on objective push.

\section*{5. Optimization Classes}
\textbf{Hierarchy:} LP $\subset$ QP ($P\succeq0$) $\subset$ QCQP
$\subset$ SOCP $\subset$ SDP.
\begin{itemize}[leftmargin=*]
  \item \textbf{SOCP:} $\|Ax+b\|_2 \le c^\top x + d$.
  \item \textbf{Trap:} $|A_i x + b_i|^2 \le c^\top x + d$ is
    \textbf{not} always QP (if $c \neq 0$, quad constr is not a
    quadratic form). It \textit{is} SOCP.
\end{itemize}

\section*{6. Smoothness, Strong Conv. \& Rates}
\textbf{Smooth ($L$):} $\nabla^2 f \preceq LI$. \textbf{Strong Conv
($\mu$):} $\nabla^2 f \succeq \mu I$.
\textbf{Condition Number:} $\kappa = L/\mu$.

\begin{table}[H]
  \footnotesize
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Setup} & \textbf{Method} & \textbf{Rate / Iters $N$} \\ \midrule
    $L, \mu$ & GD / PGD & $O(\kappa \ln(1/\epsilon))$ \\
    $L, \mu$ & Nesterov & $O(\sqrt{\kappa} \ln(1/\epsilon))$ \\
    $L$ only & GD / Nest & $O(1/\epsilon)$ / $O(1/\sqrt{\epsilon})$ \\
    Lipschitz ($L_f$) & Subgrad & $O(1/\epsilon^2)$ \\ \bottomrule
  \end{tabular}
\end{table}

\textbf{Iteration Bound Recipe:}
For $\|x_N - x^*\| \le \epsilon$ with optimal step $T = \frac{2}{L+\mu}$:
\[ N \ge \frac{\ln(|x_0 - x^*|/\epsilon)}{\ln((\kappa+1)/(\kappa-1))}
\approx \frac{\kappa+1}{2} \ln \frac{|x_0-x^*|}{\epsilon} \]

\section*{7. Hand Calculus \& Matrix Hacks}
\textbf{Derivatives:} $\nabla(x^\top A x)=(A+A^\top)x$. $\nabla^2 =
A+A^\top$. \\
\textbf{Quadratic Form:} $f = \frac{1}{2}x^\top Q x + b^\top x + c
\implies \nabla^2 f = Q$. Min at $-Q^{-1}b$ (if $Q\succ0$). $\mu =
\lambda_{min}(Q), L = \lambda_{max}(Q)$. \\
\textbf{Steepest Descent Step:} $T_k = \frac{\nabla f_k^\top \nabla
f_k}{\nabla f_k^\top Q \nabla f_k}$.

\textbf{Completing the Square (Vector):}
$x^\top Q x - 2 b^\top x = (x-Q^{-1}b)^\top Q (x-Q^{-1}b) - b^\top Q^{-1}b$.

\textbf{$2 \times 2$ Eigenvalues (Fast L \& $\mu$):} \\
$\lambda^2 - \text{Tr}(A)\lambda + \det(A) = 0 \implies \lambda_{1,2}
= \frac{\text{Tr} \pm \sqrt{\text{Tr}^2 - 4\det}}{2}$.

\textbf{$2 \times 2$ Inverse:}
$A^{-1} = \frac{1}{\det(A)}
\begin{pmatrix} d & -b \\ -c & a
\end{pmatrix}$.

\textbf{Sherman-Morrison (Rank-1 Update):} \\
$(A+uv^\top)^{-1} = A^{-1} - \frac{A^{-1}uv^\top A^{-1}}{1+v^\top A^{-1}u}$.

\textbf{Schur Complement (Solving KKT Linear Systems):}
\[
  \begin{bmatrix} Q & A^\top \\ A & 0
  \end{bmatrix}
  \begin{bmatrix} x \\ \nu
  \end{bmatrix} =
  \begin{bmatrix} -c \\ b
  \end{bmatrix}
\]
$\implies \text{Solve for } \nu: (A Q^{-1} A^\top) \nu = -(A Q^{-1} c + b)$. \\
$\implies \text{Solve for } x: x = -Q^{-1}(A^\top \nu + c)$.
