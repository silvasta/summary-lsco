\section{1. Convexity (Quiz 1 Tactic)}
\textbf{Basic Tests:}
\begin{itemize}[leftmargin=*]
  \item \textbf{Def:} $f(\theta x + \dots) \le \theta f(x) + \dots$
    $\iff$ epi$(f)$ convex.
  \item \textbf{1st-order:} $f(y) \ge f(x) + \nabla f(x)^\top(y-x)$.
  \item \textbf{2nd-order:} $\nabla^2 f(x) \succeq 0$.
    \textit{Strict} if $\succ 0$.
  \item \textbf{Operations:} Pointwise max $g(x)=\max_i f_i(x)$ is
    convex. Non-negative sum is convex. $f(Ax+b)$ is convex.
  \item \textbf{Composition:} $h(f(x))$ convex if $h$ conv/non-dec and $f$ conv.
\end{itemize}
\textbf{The "NOT CONVEX" Trap List (MCQ):}
\begin{itemize}[leftmargin=*]
  \item Pointwise \textbf{min} of convex functions.
  \item $f(x)$ has a unique global min $\nRightarrow$ convex.
  \item $\nabla^2 f$ bounded below $\nRightarrow$ convex (must be $\ge 0$).
  \item Product $f_1 \cdot f_2$ of convex $\nRightarrow$ convex.
  \item $1/f(x)$ where $f$ is convex (e.g., $1/x^2$).
  \item $f(x)=1/\ln x$ on $(1, \infty)$ (Check Hessian: $x=e^2 \to$ neg).
\end{itemize}

\section{2. Conjugates $f^*(y)$ (The Recipe)}
\textbf{Definition:} $f^*(y) = \sup_{x \in dom f} \{y^\top x - f(x)\}$.
\textbf{Step-by-Step Calculation:}
1. Find stationary pt: $\nabla_x (y^\top x - f(x)) = 0 \implies
\nabla f(x) = y$.
2. Solve for $x$ in terms of $y$ ($x_y$).
3. If $x_y \in \text{int}(dom f)$, $f^*(y) = y^\top x_y - f(x_y)$.
4. \textbf{Restricted Domain (Key Quiz Task):} If $x \in [a, b]$,
check if $x_y$ falls outside.
\begin{itemize}[leftmargin=*]
  \item If $x_y < a$, max is at $x=a$.
  \item If $x_y > b$, max is at $x=b$.
  \item Result is usually \textit{piecewise}.
\end{itemize}
\textbf{Standard Pairs:}
\begin{itemize}[leftmargin=*]
  \item $f(x) = -\ln x \to f^*(y) = -\ln(-y)-1, y<0$.
  \item $f(x) = e^x \to f^*(y) = y \ln y - y, y>0$.
  \item $f(x) = \|x\| \to f^*(y) = 0$ if $\|y\|_* \le 1$, else $\infty$.
  \item $f(x) = \frac{1}{2}x^\top Q x \to f^*(y) = \frac{1}{2}y^\top Q^{-1} y$.
\end{itemize}

\section{3. Duality \& KKT (Problem 2/4)}
\textbf{Lagrangian:} $\mathcal{L} = f(x) + \sum \lambda_i g_i(x) +
\nu^\top(Ax-b)$.
\textbf{KKT Conditions (Stationarity, Primal/Dual Feas, Comp. Slack):}
1. $\nabla_x \mathcal{L} = 0$.
2. $g_i(x) \le 0, Ax=b$.
3. $\lambda_i \ge 0$.
4. $\lambda_i g_i(x) = 0$.
\textbf{Solving Tactics:}
\begin{itemize}[leftmargin=*]
  \item Assume constraints are active/inactive based on the objective
    (e.g., $1/x$ pushes $x$ to be large).
  \item \textbf{Slater's:} $\exists x$ s.t. $g_i(x) < 0$. If convex +
    Slater $\implies$ KKT is \textit{necessary and sufficient}.
  \item \textbf{Dual Function:} $d(\lambda, \nu) = \inf_x
    \mathcal{L}$. \textit{Always} concave.
\end{itemize}

\section{4. Optimization Classes}
\textbf{Hierarchy:} LP $\subset$ QP $\subset$ QCQP $\subset$ SOCP $\subset$ SDP.
\begin{itemize}[leftmargin=*]
  \item \textbf{SOCP:} $\|Ax+b\|_2 \le c^\top x + d$.
  \item \textbf{Trick:} $|A_i x + b_i|^2 \le c^\top x + d$ is
    \textbf{not} QP if $c \neq 0$; it is SOCP.
  \item \textbf{SDP:} $X \succeq 0$. Any SOCP can be written as SDP.
\end{itemize}

\section{5. Algorithms \& Rates}
\textbf{Smoothness ($L$):} $\nabla^2 f \preceq LI$. \textbf{Strong
Conv ($\mu$):} $\nabla^2 f \succeq \mu I$.
\textbf{Condition Number:} $\kappa = L/\mu$.
\begin{table}[H]
  \footnotesize
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Setup} & \textbf{Method} & \textbf{Rate / Iterations $N$}
    \\ \midrule
    $L, \mu$ & GD & $O(\kappa \ln(1/\epsilon))$ \\
    $L, \mu$ & Nesterov & $O(\sqrt{\kappa} \ln(1/\epsilon))$ \\
    $L$ only & GD / Nest & $O(1/\epsilon)$ / $O(1/\sqrt{\epsilon})$ \\
    Lipschitz & Subgrad & $O(1/\epsilon^2)$ iterations \\ \bottomrule
  \end{tabular}
\end{table}

\textbf{Iteration Bound Recipe (2023 Quiz):}
For $\|x_N - x^*\| \le \epsilon$ with optimal step $T = \frac{2}{L+\mu}$:
\[ N \ge \frac{\ln(|x_0 - x^*|/\epsilon)}{\ln((\kappa+1)/(\kappa-1))}
\approx \frac{\kappa+1}{2} \ln \frac{|x_0-x^*|}{\epsilon} \]

\section{6. Numerical / Matrix Toolbox}
\textbf{Hessian of Quadratic:} $f = \frac{1}{2}x^\top Q x + b^\top x
+ c \implies \nabla^2 f = Q$.
\begin{itemize}[leftmargin=*]
  \item $\mu = \lambda_{min}(Q)$, $L = \lambda_{max}(Q)$.
  \item \textbf{Steepest Descent Step:} $T_k = \frac{\nabla f_k^\top
    \nabla f_k}{\nabla f_k^\top Q \nabla f_k}$.
  \item \textbf{Search Dir:} GD $\to -\nabla f_k$. Conjugate Grad
    $\to$ Orthogonal to previous steps in $Q$-norm.
\end{itemize}

\section{7. Projections (PGD)}
\textbf{Update:} $x_{k+1} = \text{prox}_{T, C}(x_k - T \nabla f(x_k))$.
\begin{itemize}[leftmargin=*]
  \item \textbf{$\ell_\infty$ ball ($\|x\|_\infty \le 1$):} $x_i =
    \text{clamp}(z_i, -1, 1)$.
  \item \textbf{$\ell_2$ ball ($\|x\|_2 \le 1$):} $z / \max(1, \|z\|_2)$.
  \item \textbf{$\ell_1$ ball ($\|x\|_1 \le 1$):} Soft-thresholding
    (Quiz 2025: projection of $(-1,-1)$ onto $\|x\|_1 \le 1$ is $(-0.5, -0.5)$).
  \item \textbf{Orthogonal Matrix $U$:} If $C = \{x \mid
    \|Ux\|_\infty \le 1\}$, use $x = U^\top \text{prox}_{\|\cdot\|_\infty}(Uz)$.
\end{itemize}

\section{8. Subdifferentials}
$0 \in \partial f(x^*) \iff$ optimal.
\begin{itemize}[leftmargin=*]
  \item $\partial |x| = \text{sgn}(x)$ if $x \neq 0$, and $[-1, 1]$ if $x=0$.
  \item \textbf{Graphical Tactic:} Minimizer is where the
    subdifferential plot crosses or contains the 0-line (usually a
    vertical segment in the $\partial f$ plot).
\end{itemize}
