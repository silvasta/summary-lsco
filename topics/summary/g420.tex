
\section*{Convexity (MCQ/TF every year)}
\textbf{Def.} \(f\) convex \(\iff\) epi\((f)\) convex \(\iff f(\theta
x+(1-\theta)y)\le\theta f(x)+(1-\theta)f(y)\ \forall\theta\in[0,1]\).

\textbf{Characterizations (use these):}
\begin{itemize}
  \item Twice diff.: \(\nabla^2 f(x)\succeq0\ \forall x\)
    (eigenvalues for quadratics \(x^\top Qx\)).
  \item 1st-order: \(f(y)\ge f(x)+\nabla f(x)^\top(y-x)\ \forall x,y\).
  \item Monotone grad: \((\nabla f(x)-\nabla f(y))^\top(x-y)\ge0\).
  \item \(f^{**}=f\) (proper, closed epi).
  \item Restriction to line convex.
\end{itemize}

\textbf{Preserving:} nonnegative lin.\ comb., pointwise max, affine
comp., intersection of sets.

\textbf{Does \emph{NOT} imply convex (traps):}
\begin{itemize}
  \item Unique/global minimizer or single local min (counterex:
    \(f(x)=|x|\) on \([-1,1]\), \(1\) outside).
  \item \(\nabla^2 f\) bounded below.
  \item \(|\nabla f(x)-\nabla f(y)|\ge\mu|x-y|\) (\(\mu>0\)).
  \item Pointwise min of convex funcs (counterex: \((x+1)^2\), \((x-1)^2\)).
  \item Product \(f_1 f_2\) (counterex: \(x\cdot(-x)=-x^2\)).
  \item \(g(x)=\exp(f(x))\) if \(f\) convex (not always).
  \item \(\sqrt{x}\) on \([0,1]\): no finite Lip.\ const.\ (\(\nabla
    f\to\infty\) at 0).
\end{itemize}

\textbf{Examples:}
\begin{itemize}
  \item \(\frac12 x^2 + p\ln x\) on \([1,\infty)\): convex \(\iff
    p\le1\) (\(\nabla^2=1-p/x^2\ge0\)).
  \item \(1/x_i\) convex on \(\{x>0\}\).
  \item \(f(x)=1/\ln x\) (\(x>1\)): check Hessian (not convex).
\end{itemize}

\textbf{Tactic (MC/TF):} Test Hessian/1st-order or give counterex.
Local min = global \(\implies\) convex? \textbf{False}.

\section*{Conjugate \(f^*(y)=\sup_x(y^\top x - f(x))\)}
\(f\) proper convex closed \(\implies f^{**}=f\). \(y\in\partial
f(x)\iff x\in\partial f^*(y)\).

\textbf{Compute:} Maximize \(yx-f(x)\); stationary pt.\ (deriv=0) if
in dom; else boundaries. Cases needed.

\textbf{Quiz Examples:}
\begin{itemize}
  \item \(f(x)=-x^2\), \(x\in[-1,1]\): \(f^*(y)=1+|y|\) (V-shape).
  \item \(f(x)=-\ln x\) (\(x>0\)): \(f^*(y)=-1-\ln(-y)\) (\(y<0\)),
    \(\infty\) else.
  \item \(f(x)=e^x\): \(f^*(y)=y\ln y - y\) (\(y>0\)), \(\infty\) else.
  \item \(f(x)=x^2+x\), \(x\in[-1,1]\):
    \[
      f^*(y)=
      \begin{cases}
        -y & y\le-1\\
        \frac14(y-1)^2 & -1\le y\le3\\
        y-2 & y\ge3
      \end{cases}
    \]
    (Stationary \(x=\frac12(y-1)\); check interval).
\end{itemize}

\textbf{Tactic:} Plot asked (label axes). Stationary often quadratic
result; evaluate boundaries.

\section*{Subdifferential \& Optimality}
\(\partial f(\bar x)=\{v\mid f(x)\ge f(\bar x)+v^\top(x-\bar x)\ \forall x\}\).

\textbf{Optimality:} \(0\in\partial f(x^*)\iff x^*\) global min.

\textbf{Example:} \(f(x)=|x|\):
\[
  \partial f(x)=
  \begin{cases}
    \{-1\} & x<0\\
    [-1,1] & x=0\\
    \{1\} & x>0
  \end{cases}
\]
Minimizer where \(0\in\partial f\) (interval includes 0).

\textbf{Tactic (plot questions):} Find point where subdiff plot contains 0.

\section*{Duality, Lagrangian, KKT}
Primal: \(\min f(x)\) s.t.\ \(g_i(x)\le0\), \(h_j(x)=0\).

Lagrangian: \(\mathcal L(x,\lambda,\nu)=f(x)+\sum\lambda_i
g_i(x)+\sum\nu_j h_j(x)\), \(\lambda\ge0\).

Dual fn: \(d(\lambda,\nu)=\inf_x\mathcal L(x,\lambda,\nu)\).

\textbf{Weak duality:} \(d^*\le p^*\) \textbf{always}.

\textbf{Strong duality + KKT suff./nec.\ (convex + Slater:
\(\exists\hat x\), \(g_i(\hat x)<0\))}:
\begin{align*}
  \nabla_x\mathcal L(x^*,\lambda^*,\nu^*)&=0\\
  g(x^*)&\le0,\ h(x^*)=0,\ \lambda^*\ge0\\
  \lambda^{*\top}g(x^*)&=0\ (\text{compl.\ slackness})
\end{align*}

\textbf{2023 SVM ex.:} \(\min\frac12\|a\|^2\) s.t.\ \(y_i(a^\top
x_i+b)\ge1\). Dual \(\max_{\lambda\ge0}d(\lambda)\). For
\(x_A=(0,0)^\top,y_A=-1\), \(x_B=(2,2)^\top,y_B=1\):
\(\lambda_A^*=\lambda_B^*=1/4\), \(a^*=(1/2,1/2)^\top\), \(b^*=-1\).

\textbf{2024 ex.:} \(f=1/x_1+4/x_2\), \(x_1+x_2\le3\),
\(x_{1,2}\ge1\): write \(\mathcal L\), solve KKT stationarity+slack
\(\to x^*\), \(p^*\).

\textbf{Tactic:} Write full \(\mathcal L\), set \(\nabla_x\mathcal
L=0\), substitute into \(d(\lambda)\), maximize (often quadratic). Check Slater.

\section*{Problem Classes (Hierarchy)}
LP \(\subset\) QP (\(P\succeq0\)) \(\subset\) QCQP \(\subset\) SOCP
\(\subset\) SDP.

\textbf{2024 Trap:} \(\min f^\top x\) s.t.\ \(|A_ix+b_i|^2\le
c_i^\top x+d_i\), \(Fx=g\):
\begin{itemize}
  \item SOCP: yes (\(\|A_ix+b_i\|_2\le c_i^\top x+d_i\)).
  \item Not always QP/SDP (quad constr.\ not quadratic form).
\end{itemize}

\section*{Smoothness, Strong Conv., Rates}
\(\mu\)-strongly convex + \(L\)-smooth \(\implies\kappa=L/\mu\).

\textbf{Defs:}
\begin{itemize}
  \item \(L\)-smooth: \(\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\)
    \(\iff\nabla^2\preceq LI\).
  \item \(\mu\)-str.\ conv.: \(f(y)\ge f(x)+\nabla
    f(x)^\top(y-x)+\frac\mu2\|x-y\|^2\) \(\iff\nabla^2\succeq\mu I\).
  \item Lip.\ convex (\(f\) Lip.\ const.\ \(L_f\)): subgrad.\ method.
\end{itemize}

\textbf{Convergence (\(f(x_N)-f(x^*)\le\varepsilon\)):}
\begin{itemize}
  \item \(\mu\)-sc, \(L\)-sm GD: \(N\sim\kappa\ln(1/\varepsilon)\).
  \item Nesterov acc.\ (\(\mu>0\)): \(N\sim\sqrt{\kappa}\ln(1/\varepsilon)\).
  \item \(L\)-sm convex (no \(\mu\)): Nesterov
    \(N\sim\sqrt{LR^2/\varepsilon}\) (\(O(1/\sqrt{\varepsilon})\)).
  \item Lip.\ convex (subgrad.): \(N\sim(RL_f/\varepsilon)^2\)
    (\(O(1/\varepsilon^2)\)).
  \item Projected GD: same rate as GD (if proj.\ cheap:
    \(\ell_1\)-ball, simplex, \(\ell_\infty\)).
\end{itemize}

\textbf{Tactic (rates):} Match assumptions exactly. \(\sqrt{x}\) on
\([0,1]\): Lip.\ const.\ does \textbf{not exist}.

\section*{Tactics \& Formulas}
\textbf{MC/TF Convexity:} Hessian or counterex.\ (use above).
\(f=f^{**}\) needs proper+closed epi.

\textbf{Conjugate/KKT:} Explicit deriv + boundary cases; compl.\ slackness key.

\textbf{Plots:} Hyperplane through midpoint (SVM); V-shape for
\(1+|y|\); label axes.

\textbf{Matrix:} \(\nabla(x^\top A x)=(A+A^\top)x\), \(\nabla^2=A+A^\top\).

\textbf{Quadratic} \(f(x)=\frac12 x^\top Qx + b^\top x\): min at
\(-Q^{-1}b\) if \(Q\succ0\).

\textbf{General Tips:}
\begin{itemize}
  \item Weak duality always. Strong + KKT under Slater.
  \item \(\partial f(0)=[-1,1]\) for \(|x|\).
  \item Check domains (e.g.\ \(x>0\) for \(\ln\)).
  \item One A4: prioritize convexity tests, conjugate examples, KKT
    steps, rate table.
\end{itemize}

\textbf{Common funcs (memorize conjugates):} \(\pm x^2\)
(restricted), \(\ln\), \(\exp\), \(1/x_i\), \(x^2+x\), norms.
